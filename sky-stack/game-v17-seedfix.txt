
// v17 seed independence fix: spawn probability and pocket position use different hashes.
function pocketDefs(ch){
  liquidInit();
  if(pocketCache.has(ch))return pocketCache.get(ch);
  const base=ch*24,defs=[];
  const had=id=>fluidSeeded.has(id)||revealedPockets.has(id);

  const surfaceId='water:surface:v14:'+ch;
  if(had(surfaceId)||hash01(ch,0,701)<.41){
    let width,left,cells=[];
    if(ch===-1&&had(surfaceId)){width=4;left=base+20}
    else if(ch===0&&had(surfaceId)){width=4;left=base}
    else{width=4+Math.floor(hash01(ch,0,702)*4);left=base+3+Math.floor(hash01(ch,0,703)*Math.max(1,18-width))}
    for(let x=left;x<left+width;x++){cells.push([x,0]);if(x>left&&x<left+width-1)cells.push([x,1])}
    defs.push({id:surfaceId,kind:'water',surface:true,cells})
  }

  const stoneAId='water:stoneA:v14:'+ch;
  if(had(stoneAId)||hash01(ch,1,718)<.50){
    const cx=base+4+Math.floor(hash01(ch,1,711)*16),cy=10+Math.floor(hash01(ch,1,712)*12);
    const rx=2+Math.floor(hash01(ch,1,713)*2),ry=1+Math.floor(hash01(ch,1,714)*2);
    defs.push({id:stoneAId,kind:'water',surface:false,cells:ellipseCells(cx,cy,rx,ry)})
  }
  const stoneBId='water:stoneB:v14:'+ch;
  if(had(stoneBId)||hash01(ch,3,715)<.29){
    const cx=base+5+Math.floor(hash01(ch,3,716)*14),cy=12+Math.floor(hash01(ch,3,717)*10);
    defs.push({id:stoneBId,kind:'water',surface:false,cells:ellipseCells(cx,cy,2,1)})
  }

  const lavaAId='lava:deepA:v14:'+ch;
  if(had(lavaAId)||hash01(ch,2,721)<.45){
    const cx=base+4+Math.floor(hash01(ch,2,722)*16),cy=34+Math.floor(hash01(ch,2,723)*13);
    const rx=2+Math.floor(hash01(ch,2,724)*2),ry=1+Math.floor(hash01(ch,2,725)*2);
    defs.push({id:lavaAId,kind:'lava',surface:false,cells:ellipseCells(cx,cy,rx,ry)})
  }
  const lavaBId='lava:deepB:v14:'+ch;
  if(had(lavaBId)||hash01(ch,4,726)<.225){
    const cx=base+5+Math.floor(hash01(ch,4,727)*14),cy=38+Math.floor(hash01(ch,4,728)*10);
    defs.push({id:lavaBId,kind:'lava',surface:false,cells:ellipseCells(cx,cy,2,1)})
  }

  pocketCache.set(ch,defs);return defs
}
