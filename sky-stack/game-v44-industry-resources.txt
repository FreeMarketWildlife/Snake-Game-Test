function oreAtV44(cx,cy,material){
  // Independent salt per resource; one payload per base cell, small vein clusters.
  const region=8,rx=Math.floor(cx/region),ry=Math.floor(cy/region);
  for(const [type,d] of Object.entries(ORE_DEFS_V44)){
    if(!d.materials.includes(material)||cy<d.min||cy>d.max)continue;
    for(let oy=-1;oy<=1;oy++)for(let ox=-1;ox<=1;ox++){
      const x=rx+ox,y=ry+oy;if(hash01(x,y,d.salt)>d.chance)continue;
      const px=x*region+1+Math.floor(hash01(x,y,d.salt+1)*6),py=y*region+1+Math.floor(hash01(x,y,d.salt+2)*6);
      const distance=Math.hypot(cx-px,cy-py);
      if(distance<=d.radius&&hash01(cx,cy,d.salt+3)<(distance<1.3?.9:.5))return{type,amount:d.amount+(type==='ironOre'&&cy>=16&&hash01(cx,cy,d.salt+4)<.25?1:0)};
    }
  }
  return null;
}
function markOreV44(z){
  if(!z.game.terrain||z.game.worldProduct)return z;
  const ore=oreAtV44(z.game.cx,z.game.cy,z.game.material);
  if(ore){z.game.resourceType=ore.type;z.game.resourceAmount=ore.amount}return z;
}
const industryMkV44=mk;
mk=function(...args){return markOreV44(industryMkV44(...args))};
for(const z of bs)markOreV44(z);
function grantResourceV44(type,amount,source={}){
  if(!RESOURCE_DEFS_V44[type]||!Number.isFinite(amount)||amount<=0)return false;
  inv[type]+=Math.floor(amount);
  if(Number.isFinite(source.x)&&Number.isFinite(source.y)){
    const now=performance.now(),last=oreBurstsV44.find(b=>b.type===type&&now-b.born<350&&Math.hypot(b.x-source.x,b.y-source.y)<B*3);
    if(last)last.amount+=amount;else{oreBurstsV44.push({type,amount,x:source.x,y:source.y,born:now});if(oreBurstsV44.length>12)oreBurstsV44.shift()}
  }
  return true;
}
const industryHarvestV44=harvest;
harvest=function(z,collector='player',miner=null){
  if(!z?.game||!bs.has(z))return 0;
  if(z.game.terrain&&!z.game.worldProduct&&z.game.material!=='bedrock'&&z.game.resourceType){
    grantResourceV44(z.game.resourceType,z.game.resourceAmount,{kind:collector,x:z.position.x,y:z.position.y});
  }
  // Base harvesting still owns material, gold, diamond, liquid and save behavior.
  return industryHarvestV44(z,collector,miner);
};

function canPayIndustryV44(cost){return Object.entries(cost).every(([type,n])=>(inv[type]||0)>=n)}
function payIndustryV44(cost){if(!canPayIndustryV44(cost))return false;for(const [type,n] of Object.entries(cost))inv[type]-=n;return true}
function buildFurnaceV44(){
  if(industryV44.built||!payIndustryV44(FURNACE_COST_V44))return false;

function startFurnaceV44(){
  const f=industryV44,r=RECIPES_V44[f.selected];
  if(!f.built||f.job||f.paused||!r||!payIndustryV44(r.inputs))return false;
  f.job={recipe:f.selected,remaining:r.duration};industryTickV44=performance.now();ui();saveGame();return true;
}
function stepIndustryV44(elapsed){
  const f=industryV44;if(!f.built||f.paused)return false;let changed=false,budget=Math.max(0,finite(elapsed));
  // Preserve the fractional frame left after a batch completes. Production time
  // stays consistent at 30/60/144 FPS without tying it to musical subdivisions.
  if(!f.job)budget=0;
  for(let batches=0;batches<64;batches++){
    if(!f.job&&f.auto){const r=RECIPES_V44[f.selected];if(payIndustryV44(r.inputs)){f.job={recipe:f.selected,remaining:r.duration};changed=true}}
    if(!f.job)break;
    const used=Math.min(f.job.remaining,budget);f.job.remaining-=used;budget-=used;
    if(f.job.remaining>1e-7)break;
    const r=RECIPES_V44[f.job.recipe];for(const [type,n] of Object.entries(r.outputs))grantResourceV44(type,n,{kind:'production'});f.job=null;changed=true;
  }
  return changed;
}
function updateIndustryV44(now){
  const elapsed=industryTickV44===null?0:Math.max(0,now-industryTickV44);industryTickV44=now;
  if(document.hidden)return;
  // No offline catch-up and no accumulated production after a suspended frame.
  const changed=stepIndustryV44(elapsed<=1500?elapsed:0);
  if(changed){ui();saveGame()}
  if(now-industryUiTickV44>100){industryUiTickV44=now;refreshIndustryV44()}
}
