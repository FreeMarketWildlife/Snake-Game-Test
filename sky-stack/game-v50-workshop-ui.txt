for(const [type,recipe] of Object.entries(JOBS_V46.workshop)){

 const size=document.createElement('small');size.textContent=d.w+' × '+d.h+' blocks';button.append(size);
 const costs=document.createElement('span');costs.className='recipeCosts';
 for(const [material,count] of Object.entries(recipe.inputs)){
  const cost=document.createElement('span');cost.className='recipeCost';cost.dataset.material=material;
  const img=document.createElement('img');img.src=M[material]?artIconV32(material):uiPixelV48(material);img.alt=labelV46(material);

function prepareWorkshopGridV50(b){
 $('buildingPause').hidden=false;$('buildingProgress').hidden=false;$('buildingStatus').hidden=false;
 const workshop=b.type==='workshop';$('buildingPanel').classList.toggle('workshopShop',workshop);workshopGridV50.hidden=!workshop;
 for(const el of [$('buildingHero'),$('buildingRecipe'),document.querySelector('label[for="buildingRecipe"]'),$('buildingIngredients'),$('buildingProduce')])el.hidden=workshop;
 if(workshop)$('buildingNote').textContent='Tap a structure to craft one. It goes straight to your Structures inventory.';
}
function refreshWorkshopGridV50(b){
 if(b.type!=='workshop')return;
 for(const [type,recipe] of Object.entries(JOBS_V46.workshop)){
  const button=$('craft-structure-'+type),active=b.job?.recipe===type,affordable=canPayIndustryV44(recipe.inputs);
  button.disabled=!!b.job||b.paused||!affordable;button.classList.toggle('crafting',active);
  button.querySelector('.recipeAction').textContent=active?'CRAFTING…':b.job?'BUSY':!affordable?'NEED MATERIALS':b.paused?'PAUSED':'CRAFT · 5s';
  for(const [material,count] of Object.entries(recipe.inputs)){
   const cost=button.querySelector('[data-material="'+material+'"]');cost.classList.toggle('missing',(inv[material]||0)<count);cost.title=labelV46(material)+': '+count+' needed · '+(inv[material]||0)+' owned';
  }
 }
 $('buildingPause').hidden=!b.job;$('buildingProgress').hidden=!b.job;$('buildingStatus').hidden=!b.job;
}
