function pickaxeBrushV39(p){
  const size=pickaxeTiersV39[pickaxeTier].size,offset=Math.floor((size-1)/2);
  return{x:Math.floor(p.x/B)-offset,y:Math.floor(p.y/B)-offset,size}
}
function pickaxeDragV39(g,now=performance.now()){
  if(!pickaxeTier||g.t!=='pick')return;
  if(!g.pickHits)g.pickHits=new Map();
  const from=g.pickLast||g.a,to=g.b,steps=Math.max(1,Math.ceil(Math.hypot(to.x-from.x,to.y-from.y)/(B*.5)));
  const cells=new Map();
  for(let i=0;i<=steps;i++){
    const r=pickaxeBrushV39({x:from.x+(to.x-from.x)*i/steps,y:from.y+(to.y-from.y)*i/steps});
    for(let y=r.y;y<r.y+r.size;y++)for(let x=r.x;x<r.x+r.size;x++)cells.set(key(x,y),{x:(x+.5)*B,y:(y+.5)*B})
  }
  // Top-to-bottom processing can expose the next dirt row during a single swing.
  const points=[...cells.values()].sort((a,b)=>a.y-b.y||a.x-b.x);
  for(const p of points){
    const z=at(p);if(!z?.game||z.game.liquid||z.game.reservoir)continue;
    if(now-(g.pickHits.get(z)??-Infinity)<220)continue;
    if(!devToolsV30.oneHit&&(z.game.material==='bedrock'||z.game.material==='stone'&&!stone||z.game.terrain&&!terrainExposed(z)))continue;
    g.pickHits.set(z,now);mine(p)
  }
  g.pickLast={...to}
}
