// --- v26 SAFE PINCH ZOOM / NATIVE-ZOOM GUARD ------------------------------------
// Prevent browser page zoom from stealing mining gestures. Two-finger input on the

function pinchMetricsV26(pair){

  return{mx:(p.x+q.x)/2,my:(p.y+q.y)/2,dist:Math.max(8,Math.hypot(q.x-p.x,q.y-p.y))}

  if(typeof cameraSpringVYV24!=='undefined')cameraSpringVYV24=0;

function pinchApplyV26(){

  const m=pinchMetricsV26(pair),ratio=m.dist/pinchStateV26.startDist;
  const nz=clamp(pinchStateV26.startZoom*ratio,MIN_ZOOM_V26,MAX_ZOOM_V26);
  const rawX=pinchStateV26.anchor.x-(m.mx-W/2)/nz;

  pinchStateV26=null;pointers.clear();gesture=null;

  if(typeof cameraDragRawYV24!=='undefined')cameraDragRawYV24=cam.y;
  if(typeof cameraReleaseV24==='function')cameraReleaseV24()

// Capture-phase handlers get first chance at a second finger, before the original
// single-pointer mining/build handlers can turn it into an accidental action.
c.addEventListener('pointerdown',e=>{
  if(e.pointerType!=='touch')return;
  pinchTouchesV26.set(e.pointerId,pinchPosV26(e));
  try{c.setPointerCapture(e.pointerId)}catch{}
  if(pinchTouchesV26.size>=2){
    if(!pinchStateV26)pinchBeginV26();
    e.preventDefault();e.stopImmediatePropagation()
  }

c.addEventListener('pointermove',e=>{
  if(e.pointerType!=='touch')return;
  if(pinchTouchesV26.has(e.pointerId))pinchTouchesV26.set(e.pointerId,pinchPosV26(e));
  if(pinchStateV26||pinchConsumedV26.has(e.pointerId)){
    pinchApplyV26();e.preventDefault();e.stopImmediatePropagation()
  }

function pinchPointerEndV26(e){
  if(e.pointerType!=='touch')return;
  const consumed=pinchConsumedV26.has(e.pointerId)||!!pinchStateV26;
  pinchTouchesV26.delete(e.pointerId);pinchConsumedV26.delete(e.pointerId);
  if(pinchStateV26&&pinchTouchesV26.size<2)pinchEndV26();
  if(consumed){e.preventDefault();e.stopImmediatePropagation()}
}
c.addEventListener('pointerup',pinchPointerEndV26,{capture:true,passive:false});
c.addEventListener('pointercancel',pinchPointerEndV26,{capture:true,passive:false});

  lastCanvasTouchEndV26=now

// TOP and DOWN reuse the existing camera targets, but always return to normal zoom.
const goBaseV26=go;
go=function(p){
  zoomTargetV26=DEFAULT_ZOOM_V26;zoomReturnActiveV26=true;
  pinchStateV26=null;pinchTouchesV26.clear();pinchConsumedV26.clear();

// v24 owns x/y spring physics. Wrap its tick only to animate zoom back to 1x when a
// camera navigation button is used.
const cameraTickBaseV26=cameraTickV24;
cameraTickV24=function(now){

    cam.z+=(zoomTargetV26-cam.z)*.16;
    if(Math.abs(zoomTargetV26-cam.z)<.0025){cam.z=zoomTargetV26;zoomReturnActiveV26=false}
