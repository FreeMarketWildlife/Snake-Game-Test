function progress(now){
  const p=topInfo(),m=p?p.m:0;$('height').textContent=m+' m';
  const el=$('stability');

  const stable=speed<.16&&spin<.025;
  if(!stable){highCandidate=null;el.classList.add('show');$('stabilityText').textContent='STABILIZING…';$('stabilityFill').style.width='0%';return}
  if(!highCandidate||highCandidate.bodyId!==p.body.id||Math.abs(highCandidate.m-m)>1){highCandidate={bodyId:p.body.id,m,start:now,x:p.body.position.x,y:p.body.position.y,angle:p.body.angle}}
  const drift=Math.hypot(p.body.position.x-highCandidate.x,p.body.position.y-highCandidate.y),rot=Math.abs(normAngle(p.body.angle-highCandidate.angle));
  if(drift>B*.18||rot>.045){highCandidate={bodyId:p.body.id,m,start:now,x:p.body.position.x,y:p.body.position.y,angle:p.body.angle}}

}
