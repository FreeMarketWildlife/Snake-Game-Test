// Developer tools deliberately change the current saved world. Switches are session-only.
var devToolsV30={oneHit:false,infiniteMiners:false,level:3};
function devNumberV30(value,max){
  if(String(value).trim()==='')return null;
  const n=Number(value);return Number.isFinite(n)&&n>=0&&n<=max?Math.floor(n):null
}
function devGoldV30(value){
  const amount=devNumberV30(value,1000000000);if(amount===null)return false;
  inv.gold=Math.min(1000000000,inv.gold+amount);ui();refreshMinerMenu();saveGame();return true
}
function devHeightV30(value){
  const height=devNumberV30(value,10000);if(height===null)return false;
  if(height>best)awardHigh(height);
  else{best=height;shape=best>=5;stone=best>=10;ui();quest()}
  highCandidate=null;$('stability').classList.remove('show');saveGame();return true
}
function devBreakV30(z,collector='player',worker=null){
  if(!z?.game||z.game.liquid||z.game.reservoir)return false;
  if(z.game.material==='bedrock'){
    const k=key(z.game.cx,z.game.cy);grid.delete(k);removedTerrain.add(k);terrainDamage.delete(k);
    bs.delete(z);World.remove(eng.world,z);wake();wakeFluids();ui();saveSoon();return true
  }
  SkyAudio.hit(z.game.material,true,1);harvest(z,collector,worker);return true
}
var devBaseMineV30=mine,devBaseMinerHitV30=minerHit;
mine=function(p){if(devToolsV30.oneHit)return devBreakV30(at(p));return devBaseMineV30(p)};
minerHit=function(q,target,beatIndex=0){if(devToolsV30.oneHit)return devBreakV30(target,'miner',q);return devBaseMinerHitV30(q,target,beatIndex)};
var devBaseUnlockedV30=unlockedMiners,devBaseAvailableV30=availableMiners,devBasePlaceV30=placeMiner;
unlockedMiners=function(){return devToolsV30.infiniteMiners?Infinity:devBaseUnlockedV30()};
availableMiners=function(){return devToolsV30.infiniteMiners?Infinity:devBaseAvailableV30()};
function devSpawnV30(p,level=devToolsV30.level){
  ensureChunkAtX(p.x);const drop=minerDropPoint(p);
  return createMinerAt(drop.x,drop.y,{level,dir:Math.random()<.5?-1:1})
}
placeMiner=function(p){
  if(!devToolsV30.infiniteMiners)return devBasePlaceV30(p);
  devSpawnV30(p);ui();SkyAudio.ui();saveSoon()
};
var devBaseUiV30=ui;
ui=function(){
  devBaseUiV30();if(devToolsV30.infiniteMiners)$('mc').textContent='∞';
  const summary=$('devSummary');if(summary)summary.textContent='Gold: '+inv.gold+' · Record: '+best+' m · Active miners: '+miners.size;
