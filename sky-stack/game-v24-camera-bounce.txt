// --- v24 ELASTIC VERTICAL CAMERA --------------------------------------------------
// Infinite horizontal movement, but vertical overscroll behaves like a real rubber band:
// drag stretches past the current top/depth with increasing resistance, then release
// carries a little momentum into an under-damped spring that rebounds and settles.
var cameraDeepCacheSizeV24=-1,cameraDeepYV24=0;
var cameraDragRawYV24=NaN,cameraDragLastYV24=NaN,cameraDragLastTV24=0,cameraDragVelocityV24=0;
var cameraSpringActiveV24=false,cameraSpringEdgeV24=0,cameraSpringVYV24=0,cameraSpringLastTV24=0;
SKY_BUILD='v24';

function cameraDeepestEverV24(){
  if(cameraDeepCacheSizeV24===removedTerrain.size)return cameraDeepYV24;
  cameraDeepCacheSizeV24=removedTerrain.size;let deepest=0;
  for(const k of removedTerrain){
    const i=k.lastIndexOf(',');if(i<0)continue;
    const cy=Number(k.slice(i+1));if(Number.isFinite(cy))deepest=Math.max(deepest,(cy+1)*B)
  }
  cameraDeepYV24=deepest;return deepest
}
function cameraHighestCurrentV24(){
  const p=topInfo();return p?finite(p.y):0
}
function cameraVerticalLimitsV24(){
  const highest=cameraHighestCurrentV24(),deepest=cameraDeepestEverV24();
  const view=HH/Math.max(.25,cam.z);
  const room=clamp(view*.22,B*5.5,B*8);
  let min=highest-room,max=deepest+room;
  if(max-min<B*7){const mid=(min+max)/2;min=mid-B*3.5;max=mid+B*3.5}
  return{min,max,highest,deepest,room}
}
function cameraRubberDistanceV24(d){
  const view=HH/Math.max(.25,cam.z);
  const maxStretch=clamp(view*.18,B*4,B*6.5),softness=B*3.15;
  return maxStretch*(1-Math.exp(-Math.max(0,d)/softness))
}
function cameraRubberFromRawV24(raw,L=cameraVerticalLimitsV24()){
  if(raw<L.min)return L.min-cameraRubberDistanceV24(L.min-raw);
  if(raw>L.max)return L.max+cameraRubberDistanceV24(raw-L.max);
  return raw
}
function cameraBeginDragV24(){
  cameraSpringActiveV24=false;cameraSpringEdgeV24=0;cameraSpringVYV24=0;cam.anim=false;
  cameraDragRawYV24=cam.y;cameraDragLastYV24=cam.y;cameraDragLastTV24=performance.now();cameraDragVelocityV24=0
}
function cameraStartSpringV24(forceEdge=0){
  const L=cameraVerticalLimitsV24();
  let edge=forceEdge;
  if(!edge){if(cam.y<L.min)edge=-1;else if(cam.y>L.max)edge=1;else return false}
  cameraSpringEdgeV24=edge;cameraSpringActiveV24=true;cam.anim=false;
  cameraSpringVYV24=clamp(cameraDragVelocityV24,-B*.72,B*.72);
  cameraSpringLastTV24=performance.now();return true
}
function cameraReleaseV24(){
  const L=cameraVerticalLimitsV24();
  const edge=cam.y<L.min?-1:cam.y>L.max?1:0;
  cameraDragRawYV24=NaN;cameraDragLastYV24=NaN;
  if(edge)cameraStartSpringV24(edge)
}

// Raw drag position keeps accumulating even after the visible camera enters rubber-band
// territory. That makes pulling farther feel progressively heavier instead of wall-like.
function pan(dx,dy){
  const now=performance.now();cam.anim=false;cameraSpringActiveV24=false;
  cam.x-=dx/cam.z; // deliberately unbounded forever
  if(!Number.isFinite(cameraDragRawYV24))cameraBeginDragV24();
  cameraDragRawYV24-=dy/cam.z;
  const next=cameraRubberFromRawV24(cameraDragRawYV24),dt=clamp(now-cameraDragLastTV24,6,50)/16.67;
  const inst=(next-cameraDragLastYV24)/Math.max(.35,dt);
  cameraDragVelocityV24=cameraDragVelocityV24*.58+inst*.42;
  cam.y=next;cameraDragLastYV24=next;cameraDragLastTV24=now
}
function go(p){
  const L=cameraVerticalLimitsV24();cameraSpringActiveV24=false;cameraSpringVYV24=0;
  cam.tx=p.x;cam.ty=clamp(p.y,L.min,L.max);cam.anim=true
}
function cameraSpringTickV24(now,L){
  const target=cameraSpringEdgeV24<0?L.min:L.max;
  const dt=clamp((now-(cameraSpringLastTV24||now))/16.67,.45,1.8);cameraSpringLastTV24=now;
  const delta=target-cam.y;
  // Intentionally under-damped: crosses the boundary, rebounds, then settles.
  const stiffness=.125,damping=.79;
  cameraSpringVYV24+=delta*stiffness*dt;
  cameraSpringVYV24*=Math.pow(damping,dt);
  cam.y+=cameraSpringVYV24*dt;
  if(Math.abs(target-cam.y)<.28&&Math.abs(cameraSpringVYV24)<.16){
    cam.y=target;cameraSpringVYV24=0;cameraSpringActiveV24=false;cameraSpringEdgeV24=0
  }
}
function cameraTickV24(now){
  const L=cameraVerticalLimitsV24();
  if(cam.anim){
    cameraSpringActiveV24=false;cam.ty=clamp(cam.ty,L.min,L.max);
    cam.x+=(cam.tx-cam.x)*.12;cam.y+=(cam.ty-cam.y)*.16;
    if(Math.abs(cam.tx-cam.x)<.3&&Math.abs(cam.ty-cam.y)<.3){cam.x=cam.tx;cam.y=cam.ty;cam.anim=false}
    return
  }
  if(gesture)return;
  if(!cameraSpringActiveV24){
    if(cam.y<L.min){cameraDragVelocityV24=0;cameraStartSpringV24(-1)}
    else if(cam.y>L.max){cameraDragVelocityV24=0;cameraStartSpringV24(1)}
  }
  if(cameraSpringActiveV24)cameraSpringTickV24(now,L)
}

// These listeners run alongside the existing game gesture handlers. Pointerdown seeds the
// hidden raw drag coordinate; pointerup/cancel releases the spring after the game action.
c.addEventListener('pointerdown',cameraBeginDragV24,{passive:true});
c.addEventListener('pointerup',cameraReleaseV24,{passive:true});
c.addEventListener('pointercancel',cameraReleaseV24,{passive:true});

function loopV24(now){
  const t=now||performance.now();
  Engine.update(eng,16.67);updateMiners(t);cameraTickV24(t);ensureWorld();progress(t);
  ctx.setTransform(DPR,0,0,DPR,0,0);bg();for(const z of bs)draw(z);drawMappedLiquids(t);
  for(const q of miners)drawMiner(q,t);drawGoldBursts(t);preview();requestAnimationFrame(loop)
}
loop=loopV24;

window.__skyStackCameraDebug=()=>({
  build:'v24',x:+cam.x.toFixed(1),y:+cam.y.toFixed(1),spring:cameraSpringActiveV24,
  springVY:+cameraSpringVYV24.toFixed(2),dragVY:+cameraDragVelocityV24.toFixed(2),...cameraVerticalLimitsV24()
});
