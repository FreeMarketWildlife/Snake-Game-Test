
// --- v23 VERTICAL CAMERA SOFT BOUNDS ---------------------------------------------
// Horizontal exploration remains infinite. Vertical dragging gets rubber-band resistance
// beyond the current highest placed block / deepest terrain ever mined, then springs back.
var cameraDeepCacheSizeV23=-1,cameraDeepYV23=0;
SKY_BUILD='v23';

function cameraDeepestEverV23(){
  if(cameraDeepCacheSizeV23===removedTerrain.size)return cameraDeepYV23;
  cameraDeepCacheSizeV23=removedTerrain.size;let deepest=0;
  for(const k of removedTerrain){
    const i=k.lastIndexOf(',');if(i<0)continue;
    const cy=Number(k.slice(i+1));if(Number.isFinite(cy))deepest=Math.max(deepest,(cy+1)*B)
  }
  cameraDeepYV23=deepest;return deepest
}
function cameraHighestCurrentV23(){
  const p=topInfo();return p?finite(p.y):0
}
function cameraVerticalLimitsV23(){
  const highest=cameraHighestCurrentV23(),deepest=cameraDeepestEverV23();
  // Breathing room lets the player see some sky above a tower and some rock below the
  // excavation frontier without being able to disappear into empty vertical space.
  const view=HH/Math.max(.25,cam.z);
  const room=clamp(view*.22,B*5.5,B*8);
  let min=highest-room,max=deepest+room;
  if(max-min<B*7){const mid=(min+max)/2;min=mid-B*3.5;max=mid+B*3.5}
  return{min,max,highest,deepest,room}
}
function cameraRubberV23(y){
  const L=cameraVerticalLimitsV23();
  if(y<L.min){const d=L.min-y;return L.min-(d/(1+d/(B*4)))*.34}
  if(y>L.max){const d=y-L.max;return L.max+(d/(1+d/(B*4)))*.34}
  return y
}
function cameraBounceV23(){
  const L=cameraVerticalLimitsV23(),target=clamp(cam.y,L.min,L.max);
  if(Math.abs(target-cam.y)>.15){cam.tx=cam.x;cam.ty=target;cam.anim=true}
}

// Replace the original pan: x is intentionally never clamped.
function pan(dx,dy){
  cam.anim=false;
  cam.x-=dx/cam.z;
  cam.y=cameraRubberV23(cam.y-dy/cam.z)
}
// TOP / DOWN still animate normally, but can never target forbidden vertical space.
function go(p){
  const L=cameraVerticalLimitsV23();cam.tx=p.x;cam.ty=clamp(p.y,L.min,L.max);cam.anim=true
}
function cameraTickV23(){
  const L=cameraVerticalLimitsV23();
  if(cam.anim){
    cam.ty=clamp(cam.ty,L.min,L.max);
    cam.x+=(cam.tx-cam.x)*.12;
    cam.y+=(cam.ty-cam.y)*.16;
    if(Math.abs(cam.tx-cam.x)<.3&&Math.abs(cam.ty-cam.y)<.3){cam.x=cam.tx;cam.y=cam.ty;cam.anim=false}
    return
  }
  if(gesture)return;
  const target=clamp(cam.y,L.min,L.max);
  if(Math.abs(target-cam.y)>.15){
    cam.y+=(target-cam.y)*.18;
    if(Math.abs(target-cam.y)<.35)cam.y=target
  }
}

// Existing pointerup handler performs the game action first; this second handler only
// checks whether the released camera needs to spring back inside its vertical range.
c.addEventListener('pointerup',cameraBounceV23,{passive:true});
c.addEventListener('pointercancel',cameraBounceV23,{passive:true});

// Override the hoisted main loop so dynamic bounds also react when a tower falls and its
// current highest block changes while the player is not touching the screen.
function loopV23(now){
  const t=now||performance.now();
  Engine.update(eng,16.67);updateMiners(t);cameraTickV23();ensureWorld();progress(t);
  ctx.setTransform(DPR,0,0,DPR,0,0);bg();for(const z of bs)draw(z);drawMappedLiquids(t);
  for(const q of miners)drawMiner(q,t);drawGoldBursts(t);preview();requestAnimationFrame(loop)
}
loop=loopV23;

window.__skyStackCameraDebug=()=>({build:'v23',x:+cam.x.toFixed(1),y:+cam.y.toFixed(1),...cameraVerticalLimitsV23()});
