// Volcanic glass: dark plum facets, chipped lavender edges, and a slow specular glint.
// Only the material's appearance changes; collision, mining and saves stay intact.
M.obsidian.col=['#302039','#130f20'];
var obsidianBaseDrawV31=draw;
function drawObsidianV31(z,now){
  const s=w2s(z.position.x,z.position.y),sw=z.game.w*cam.z,sh=z.game.h*cam.z;
  const radius=Math.hypot(sw,sh)/2;if(s.x+radius<0||s.x-radius>W||s.y+radius<0||s.y-radius>HH)return;
  const cx=z.game.cx??Math.floor(z.position.x/B),cy=z.game.cy??Math.floor(z.position.y/B);
  const variant=((cx*7+cy*11)%4+4)%4;
  ctx.save();ctx.translate(s.x,s.y);ctx.rotate(z.angle);
  ctx.save();ctx.scale(sw/32,sh/32);ctx.translate(-16,-16);
  ctx.beginPath();ctx.rect(0,0,32,32);ctx.clip();
  const pixel=(x,y,w,h,color)=>{ctx.fillStyle=color;ctx.fillRect(x,y,w,h)};
  pixel(0,0,32,32,'#171322');
  pixel(2,2,28,27,'#35233f');
  // Broad stepped fracture planes read as glass even when the glint is absent.
  pixel(3,3,15,8,'#54355f');pixel(3,11,10,5,'#493052');
  pixel(13,10,12,8,'#2b1b36');pixel(17,18,12,10,'#24192e');
  pixel(3,18,6,9,'#251b32');pixel(9,23,8,6,'#40284b');
  pixel(25,4,4,12,'#45294e');
  // Chipped top/left bevels separate black glass from neighboring charcoal rock.
  pixel(2,1,18,1,'#bf8cd1');pixel(2,2,11,1,'#866093');
  pixel(21,2,8,1,'#9564a9');pixel(1,3,1,16,'#9568a9');
  pixel(2,20,1,8,'#674579');pixel(3,29,24,1,'#573462');
  pixel(29,17,1,12,'#0e1020');pixel(4,30,25,1,'#100e1a');
  // A broken diagonal reflection, with stable variation between world tiles.
  const offset=variant%2;
  for(let i=0;i<4;i++){
    pixel(5+i*2+offset,5+i*3,2,3,i<2?'#a87db8':'#785388');
    pixel(7+i*2+offset,6+i*3,1,2,'#60446e');
  }
  pixel(21-offset,22,5,1,'#80528e');pixel(25-offset,20,1,2,'#a070ad');
  // Brief, staggered white-lilac glints; no random flicker or continuous flashing.
  const phase=((now/1000+variant*1.7+((cx+cy)%5+5)%5*.31)%7.5)/7.5;
  const strength=phase<.16?Math.sin(phase/.16*Math.PI):0;
  if(strength>.15){
    ctx.globalAlpha=Math.ceil(strength*4)/4;
    const x=8+offset,y=8;pixel(x,y,2,2,'#fff0ff');
    if(strength>.65){pixel(x-2,y,2,1,'#d6b4eb');pixel(x+2,y+1,2,1,'#d6b4eb');pixel(x+1,y-2,1,2,'#d6b4eb')}
    ctx.globalAlpha=1;
  }
  ctx.restore();
  // Preserve the game's damage and exposure feedback above the material artwork.
  cracks(z,sw,sh);
  if(z.game.terrain&&!terrainExposed(z)){ctx.fillStyle='rgba(7,11,16,.24)';ctx.fillRect(-sw/2,-sh/2,sw,sh)}
  if(z.game.flash>0){ctx.strokeStyle='#fff3a0';ctx.lineWidth=3;ctx.strokeRect(-sw/2+2,-sh/2+2,sw-4,sh-4);z.game.flash--}
  ctx.restore()
}
draw=function(z){
  if(z.game?.material==='obsidian')return drawObsidianV31(z,performance.now());
  return obsidianBaseDrawV31(z)
};
