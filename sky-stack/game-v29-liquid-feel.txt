function liquidTransferV29(src,x,y,amount,now){
  if(amount<=1e-9||!liquidSubOpenV22(x,y))return 0;
  const dst=liquidSubRecV22(x,y);
  if(dst&&dst.kind!==src.kind)return 0;
  amount=Math.min(src.a,amount);if(amount<=1e-9)return 0;
  src.a-=amount;
  if(dst)dst.a+=amount;
  else liquidSubV22.set(liquidSubKeyV22(x,y),{x,y,kind:src.kind,a:amount});
  liquidMarkFlowV22(src,x,y,amount,now);
  if(src.a<=1e-12)liquidSubV22.delete(liquidSubKeyV22(src.x,src.y));
  return amount
}

liquidStepSubV22=function(kind,now){
  liquidV22Init();
  const center=cam.x/liquidSBV22(),range=24*liquidSubScaleV22*3.15;
  const active=()=>[...liquidSubV22.values()].filter(r=>r.kind===kind&&Math.abs(r.x-center)<=range&&liquidSubOpenV22(r.x,r.y));
  const lava=kind==='lava',compress=lava?.012:.025;
  // Bottom first: incoming liquid cannot cascade through many cells in one tick.
  for(const r of active().sort((a,b)=>b.y-a.y||a.x-b.x)){
    const b=liquidSubRecV22(r.x,r.y+1);
    if(b&&b.kind!==kind)continue;
    const mass=b?b.a:0,total=r.a+mass;
    const target=total<=1?1:total<2+compress?(1+total*compress)/(1+compress):(total+compress)/2;
    liquidTransferV29(r,r.x,r.y+1,Math.min(lava?.24:.95,Math.max(0,target-mass)),now)
  }
  // Disjoint pairs equalize once per pass, with alternating parity to avoid bias.
  // Unsupported water falls in a column; it spreads once it reaches a pool/floor.
  for(let pass=0;pass<2;pass++){
    const pairs=new Set(),parity=(liquidStepsV29+pass)&1;
    for(const r of active()){
      const x=((r.x&1)===parity)?r.x:r.x-1,k=liquidSubKeyV22(x,r.y);
      if(pairs.has(k))continue;pairs.add(k);
      if(!liquidSubOpenV22(x,r.y)||!liquidSubOpenV22(x+1,r.y))continue;
      const a=liquidSubRecV22(x,r.y),b=liquidSubRecV22(x+1,r.y);
      if((a&&a.kind!==kind)||(b&&b.kind!==kind))continue;
      const diff=(a?a.a:0)-(b?b.a:0),src=diff>0?a:b;if(!src)continue;
      const below=liquidSubRecV22(src.x,src.y+1);
      if(liquidSubOpenV22(src.x,src.y+1)&&(!below||(below.kind===kind&&below.a<.98)))continue;
      liquidTransferV29(src,diff>0?x+1:x,r.y,Math.abs(diff)*(lava?.13:.48),now)
    }
  }
  // Small compression lets connected vessels level through submerged passages.
  for(const r of active().sort((a,b)=>a.y-b.y||a.x-b.x)){
    const up=liquidSubRecV22(r.x,r.y-1);if(up&&up.kind!==kind)continue;
    const um=up?up.a:0,total=r.a+um;
    const target=total<=1?1:total<2+compress?(1+total*compress)/(1+compress):(total+compress)/2;
    liquidTransferV29(r,r.x,r.y-1,Math.min(lava?.06:.3,Math.max(0,r.a-target)*.5),now)
  }
};

updateLiquids=function(now){
  liquidV22Init();
  if(!Number.isFinite(now))return;
  if(liquidClockV29===null){liquidClockV29=now;return}
  // Ignore time spent in a hidden tab, and bound catch-up work after a stall.
  const elapsed=now-liquidClockV29;liquidClockV29=now;
  if(elapsed<0||elapsed>250){liquidAccumulatorV29=0;return}
  liquidAccumulatorV29+=elapsed;
  const step=1000/60;
  while(liquidAccumulatorV29+1e-7>=step){
    liquidAccumulatorV29-=step;liquidStepsV29++;liquidSubTickV22++;
    const t=now-liquidAccumulatorV29;
    liquidStepSubV22('water',t);
    if(liquidStepsV29%3===0)liquidStepSubV22('lava',t);
    liquidReactV22(t);
    if(liquidStepsV29%2===0)updateBuoyancyV22();
    if(liquidStepsV29%120===0){
      // Do not evaporate shallow films: all positive volume stays in the solver.
      for(const [k,r] of liquidSubV22)if(!Number.isFinite(r.a)||r.a<=0)liquidSubV22.delete(k);
      for(const [k,q] of liquidSubFlowV22)if(t-q.last>900)liquidSubFlowV22.delete(k);
      for(const [k,q] of liquidSubImpactV22)if(t-q.last>1000)liquidSubImpactV22.delete(k);
      for(const k of liquidSubVisualV22.keys())if(!liquidSubV22.has(k.slice(k.indexOf(':')+1)))liquidSubVisualV22.delete(k)
    }
  }
};

function liquidGeometryV29(r,now){
  const sb=liquidSBV22(),above=liquidSubRecV22(r.x,r.y-1),below=liquidSubRecV22(r.x,r.y+1);
  const joined=above&&above.kind===r.kind&&above.a>.01;
  const falling=liquidSubOpenV22(r.x,r.y+1)&&(!below||(below.kind===r.kind&&below.a<.92));
  let a=liquidVisualAmountV22(r,now),top=(r.y+1-a)*sb,left=r.x*sb,width=sb;
  if(falling){
    const flow=liquidFlowAtV22(r.x,r.y,r.kind,now);
    width=Math.max(2,Math.min(sb,sb*Math.sqrt(Math.max(r.a,flow?.mag||0))));
    // Straight, connected columns instead of diagonal, sine-wobbling ribbons.
    left+=(sb-width)/2;
    // A trace left behind by a receding pool is a shallow film, not a tall spike.
    top=r.a<.04&&!joined?(r.y+1-Math.sqrt(r.a))*sb:r.y*sb;
  }else if(joined)top=r.y*sb;
  else{
    let ripple=0;
    for(let dx=-3;dx<=3;dx++){
      const impact=liquidImpactAtV22(r.x+dx,r.y,r.kind,now);
      ripple+=Math.sin(now*.014-Math.abs(dx)*1.5)*impact*(1-Math.abs(dx)/4)
    }
    top+=clamp(ripple,-1,1)*(r.kind==='lava'?.6:1.5);
    top=clamp(top,r.y*sb,(r.y+1)*sb-1);
  }
  return{left,top,width,bottom:(r.y+1)*sb,surface:!joined&&!falling,falling}
}

drawMappedLiquids=function(now){

  const top=Math.floor((cam.y-HH/(2*cam.z))/pixel)*pixel;

  if(!liquidCanvasV29){liquidCanvasV29=document.createElement('canvas');liquidContextV29=liquidCanvasV29.getContext('2d')}

  }
  // Opaque offscreen shapes, composited once: no dark overlaps or tile seams.
  for(const {r,shape:s} of visible){

    else rect(s.left,s.top,s.width,s.bottom-s.top,color);

  }
